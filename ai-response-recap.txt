                Icon(

                )

        }

                .fillMaxWidth()
                .horizontalScroll(rememberScrollState())

                            contentDescription = null,
                            modifier = Modifier
                                .size(InputChipDefaults.IconSize)
                                .clickable { onDeleteClick(history.id) },

                    },
                )
            }
        }
    }
